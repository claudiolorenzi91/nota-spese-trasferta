    <script>
        const currencySymbols = {
            'EUR': '€',
            'USD': '$',
            'GBP': '£',
            'CHF': 'Fr',
            'JPY': '¥',
            'CAD': '$',
            'AUD': '$',
            'CNY': '¥'
        };

        let expenses = [];
        let tripDetails = {
            firstName: "Mario",
            lastName: "Rossi",
            companyName: "Azienda S.p.A.",
            location: "Italia, Milano"
        };

        window.addEventListener('DOMContentLoaded', () => {
            const savedExpenses = localStorage.getItem('corporate_expenses');
            if (savedExpenses) {
                try {
                    expenses = JSON.parse(savedExpenses);
                } catch(e) {
                    console.error("Errore nel parsing delle spese salvate", e);
                }
            }

            const savedTrip = localStorage.getItem('corporate_trip_details');
            if (savedTrip) {
                try {
                    const parsed = JSON.parse(savedTrip);
                    // Supporta retrocompatibilità se salvato come travelerName unico
                    if (parsed.travelerName && !parsed.firstName) {
                        const parts = parsed.travelerName.split(' ');
                        tripDetails.firstName = parts[0] || 'Mario';
                        tripDetails.lastName = parts.slice(1).join(' ') || 'Rossi';
                    } else {
                        tripDetails.firstName = parsed.firstName || "Mario";
                        tripDetails.lastName = parsed.lastName || "Rossi";
                    }
                    tripDetails.companyName = parsed.companyName || "Azienda S.p.A.";
                    tripDetails.location = parsed.location || "Italia, Milano";
                } catch(e) {
                    console.error("Errore nel parsing dei dettagli trasferta", e);
                }
            }

            updateTripDetailsUI();

            const today = new Date().toISOString().split('T')[0];
            document.getElementById('date').value = today;

            if (expenses.length > 0) {
                const dates = expenses.map(e => e.date).sort();
                document.getElementById('filterStartDate').value = dates[0];
                document.getElementById('filterEndDate').value = dates[dates.length - 1];
            } else {
                document.getElementById('filterStartDate').value = today;
                document.getElementById('filterEndDate').value = today;
            }

            renderExpenses();
        });

        function saveToLocalStorage() {
            localStorage.setItem('corporate_expenses', JSON.stringify(expenses));
            localStorage.setItem('corporate_trip_details', JSON.stringify(tripDetails));
        }

        function updateTripDetailsUI() {
            document.getElementById('displayFirstName').textContent = tripDetails.firstName;
            document.getElementById('displayLastName').textContent = tripDetails.lastName;
            document.getElementById('displayCompanyName').textContent = tripDetails.companyName;
            document.getElementById('displayLocation').textContent = tripDetails.location;
            document.getElementById('headerTripSummaryBtn').textContent = `${tripDetails.firstName} ${tripDetails.lastName} - ${tripDetails.location}`;
        }

        function openTripSettingsModal() {
            document.getElementById('inputFirstName').value = tripDetails.firstName;
            document.getElementById('inputLastName').value = tripDetails.lastName;
            document.getElementById('inputCompanyName').value = tripDetails.companyName;
            document.getElementById('inputLocation').value = tripDetails.location;
            document.getElementById('tripSettingsModal').classList.remove('hidden');
        }

        function closeTripSettingsModal() {
            document.getElementById('tripSettingsModal').classList.add('hidden');
        }

        function saveTripSettings(e) {
            e.preventDefault();
            tripDetails.firstName = document.getElementById('inputFirstName').value.trim();
            tripDetails.lastName = document.getElementById('inputLastName').value.trim();
            tripDetails.companyName = document.getElementById('inputCompanyName').value.trim();
            tripDetails.location = document.getElementById('inputLocation').value.trim();
            
            saveToLocalStorage();
            updateTripDetailsUI();
            closeTripSettingsModal();
        }

        function previewSelectedFile(input) {
            const container = document.getElementById('formImagePreviewContainer');
            const thumbnail = document.getElementById('formImageThumbnail');
            const nameSpan = document.getElementById('formImageName');
            const sizeSpan = document.getElementById('formImageSize');

            if (input.files && input.files[0]) {
                const file = input.files[0];
                const fileSizeMB = file.size / (1024 * 1024);

                if (fileSizeMB > 50) {
                    alert('Il file selezionato supera i 50 MB consentiti. Si prega di scegliere un file più leggero.');
                    input.value = '';
                    container.classList.add('hidden');
                    return;
                }

                const reader = new FileReader();
                reader.onload = function(e) {
                    thumbnail.src = e.target.result;
                    nameSpan.textContent = file.name;
                    sizeSpan.textContent = `${fileSizeMB.toFixed(2)} MB`;
                    container.classList.remove('hidden');
                }
                reader.readAsDataURL(file);
            }
        }

        function clearFileSelection() {
            const input = document.getElementById('receiptImage');
            input.value = '';
            document.getElementById('formImagePreviewContainer').classList.add('hidden');
        }

        function editExpense(id) {
            const expense = expenses.find(e => e.id === id);
            if (!expense) return;

            document.getElementById('editingExpenseId').value = expense.id;
            document.getElementById('date').value = expense.date;
            document.getElementById('description').value = expense.description;
            document.getElementById('amount').value = expense.amount;
            document.getElementById('currency').value = expense.currency || 'EUR';

            const container = document.getElementById('formImagePreviewContainer');
            const thumbnail = document.getElementById('formImageThumbnail');
            const nameSpan = document.getElementById('formImageName');
            const sizeSpan = document.getElementById('formImageSize');

            thumbnail.src = expense.imageSrc;
            nameSpan.textContent = expense.fileName || 'scontrino.jpg';
            sizeSpan.textContent = 'Immagine allegata';
            container.classList.remove('hidden');

            document.getElementById('formTitle').innerHTML = '<i class="fa-solid fa-pen-to-square text-brand-600"></i> Modifica Spesa';
            document.getElementById('submitExpenseBtnText').textContent = 'Aggiorna Spesa';
            document.getElementById('cancelEditBtn').classList.remove('hidden');

            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function cancelEditing() {
            document.getElementById('editingExpenseId').value = '';
            document.getElementById('expenseForm').reset();
            clearFileSelection();

            document.getElementById('formTitle').innerHTML = '<i class="fa-solid fa-circle-plus text-brand-600"></i> Nuova Spesa';
            document.getElementById('submitExpenseBtnText').textContent = 'Salva Spesa';
            document.getElementById('cancelEditBtn').classList.add('hidden');

            const today = new Date().toISOString().split('T')[0];
            document.getElementById('date').value = today;
        }

        document.getElementById('expenseForm').addEventListener('submit', function(e) {
            e.preventDefault();

            const editingId = document.getElementById('editingExpenseId').value;
            const date = document.getElementById('date').value;
            const description = document.getElementById('description').value;
            const currency = document.getElementById('currency').value;
            const amount = parseFloat(document.getElementById('amount').value);
            const fileInput = document.getElementById('receiptImage');

            const existingExpense = editingId ? expenses.find(e => e.id === editingId) : null;

            if (!fileInput.files || !fileInput.files[0]) {
                if (existingExpense) {
                    existingExpense.date = date;
                    existingExpense.description = description;
                    existingExpense.currency = currency;
                    existingExpense.amount = amount;

                    saveToLocalStorage();
                    renderExpenses();
                    cancelEditing();
                    return;
                } else {
                    alert('Si prega di caricare la foto dello scontrino.');
                    return;
                }
            }

            const file = fileInput.files[0];
            const reader = new FileReader();

            reader.onload = function(event) {
                const img = new Image();
                img.src = event.target.result;
                img.onload = function() {
                    const canvas = document.createElement('canvas');
                    let width = img.width;
                    let height = img.height;
                    const maxDim = 2000;

                    if (width > maxDim || height > maxDim) {
                        if (width > height) {
                            height = Math.round((height * maxDim) / width);
                            width = maxDim;
                        } else {
                            width = Math.round((width * maxDim) / height);
                            height = maxDim;
                        }
                    }

                    canvas.width = width;
                    canvas.height = height;
                    const ctx = canvas.getContext('2d');
                    ctx.drawImage(img, 0, 0, width, height);

                    const optimizedDataUrl = canvas.toDataURL('image/jpeg', 0.85);

                    if (editingId && existingExpense) {
                        existingExpense.date = date;
                        existingExpense.description = description;
                        existingExpense.currency = currency;
                        existingExpense.amount = amount;
                        existingExpense.imageSrc = optimizedDataUrl;
                        existingExpense.fileName = file.name;
                    } else {
                        const newExpense = {
                            id: 'exp_' + Date.now(),
                            date,
                            description,
                            currency,
                            amount,
                            imageSrc: optimizedDataUrl,
                            fileName: file.name
                        };
                        expenses.unshift(newExpense);
                    }

                    saveToLocalStorage();
                    renderExpenses();
                    cancelEditing();
                };
            };

            reader.readAsDataURL(file);
        });

        function getFilteredExpenses() {
            const startDate = document.getElementById('filterStartDate').value;
            const endDate = document.getElementById('filterEndDate').value;

            return expenses.filter(exp => {
                if (startDate && exp.date < startDate) return false;
                if (endDate && exp.date > endDate) return false;
                return true;
            });
        }

        // Genera PDF con la tabella delle spese filtrate
        function generateTableReportPDF() {
            const filtered = getFilteredExpenses();
            if (filtered.length === 0) {
                alert('Nessuna spesa trovata nell\'intervallo di date selezionato per generare il report.');
                return;
            }

            const { jsPDF } = window.jspdf;
            const doc = new jsPDF({
                orientation: 'portrait',
                unit: 'mm',
                format: 'a4'
            });

            const startDate = document.getElementById('filterStartDate').value || 'Inizio';
            const endDate = document.getElementById('filterEndDate').value || 'Fine';

            doc.setFontSize(18);
            doc.setTextColor(30, 41, 59);
            doc.text('NOTA SPESE TRASFERTA', 14, 20);

            doc.setFontSize(10, 'normal');
            doc.setTextColor(100, 116, 139);
            doc.text(`Nome: ${tripDetails.firstName}`, 14, 26);
            doc.text(`Cognome: ${tripDetails.lastName}`, 14, 32);
            doc.text(`Azienda: ${tripDetails.companyName}`, 14, 38);
            doc.text(`Luogo: ${tripDetails.location}`, 14, 44);
            doc.text(`Periodo: ${startDate} al ${endDate}`, 14, 50);

            const sortedFiltered = [...filtered].sort((a, b) => a.date.localeCompare(b.date));
            
            const totalsByCurrency = {};
            const tableRows = sortedFiltered.map((exp, index) => {
                const cur = exp.currency || 'EUR';
                const amountNum = parseFloat(exp.amount);
                if (!totalsByCurrency[cur]) totalsByCurrency[cur] = 0;
                totalsByCurrency[cur] += amountNum;

                const formattedDate = new Date(exp.date).toLocaleDateString('it-IT');
                const sym = currencySymbols[cur] || cur;
                return [
                    index + 1,
                    formattedDate,
                    exp.description,
                    `${cur}`,
                    `${sym} ${amountNum.toFixed(2)}`
                ];
            });

            // Riga riassuntiva totali per valuta
            const totalSummaryStr = Object.keys(totalsByCurrency).map(cur => {
                const sym = currencySymbols[cur] || cur;
                return `${sym} ${totalsByCurrency[cur].toFixed(2)} (${cur})`;
            }).join(' | ');

            tableRows.push([
                '',
                '',
                'TOTALE COMPLESSIVO',
                '',
                totalSummaryStr
            ]);

            doc.autoTable({
                startY: 56,
                head: [['N°', 'Data', 'Descrizione', 'Valuta', 'Importo']],
                body: tableRows,
                theme: 'striped',
                headStyles: { fillColor: [37, 99, 235], textColor: 255, fontStyle: 'bold' },
                footStyles: { fillColor: [241, 245, 249], textColor: [15, 23, 42], fontStyle: 'bold' },
                columnStyles: {
                    0: { cellWidth: 12, halign: 'center' },
                    1: { cellWidth: 28, halign: 'center' },
                    2: { cellWidth: 85 },
                    3: { cellWidth: 20, halign: 'center' },
                    4: { cellWidth: 45, halign: 'right' }
                },
                didParseCell: function(data) {
                    if (data.row.index === tableRows.length - 1) {
                        data.cell.styles.fontStyle = 'bold';
                        data.cell.styles.fillColor = [226, 232, 240];
                    }
                }
            });

            const cleanStart = startDate !== 'Inizio' ? startDate.replace(/-/g, '') : 'tutti';
            const cleanEnd = endDate !== 'Fine' ? endDate.replace(/-/g, '') : 'tutti';
            doc.save(`nota_spese_${tripDetails.lastName}_${tripDetails.firstName}_${cleanStart}_${cleanEnd}.pdf`);
        }

        // Genera PDF scontrino singolo con intestazione data, descrizione e importo con valuta
        function generateReceiptPDF(id) {
            const expense = expenses.find(e => e.id === id);
            if (!expense) return;

            const { jsPDF } = window.jspdf;
            const doc = new jsPDF({
                orientation: 'portrait',
                unit: 'mm',
                format: 'a4'
            });

            const img = new Image();
            img.src = expense.imageSrc;
            img.onload = function() {
                const pageWidth = doc.internal.pageSize.getWidth();
                const pageHeight = doc.internal.pageSize.getHeight();

                const cur = expense.currency || 'EUR';
                const sym = currencySymbols[cur] || cur;

                doc.setFontSize(14);
                doc.setTextColor(30, 41, 59);
                doc.text('SCONTRINO SPESA TRASFERTA', 14, 15);

                doc.setFontSize(10, 'normal');
                doc.setTextColor(100, 116, 139);
                doc.text(`Nome: ${tripDetails.firstName} | Cognome: ${tripDetails.lastName}`, 14, 22);
                doc.text(`Azienda: ${tripDetails.companyName}`, 14, 28);
                doc.text(`Data: ${new Date(expense.date).toLocaleDateString('it-IT')} | Importo: ${sym} ${parseFloat(expense.amount).toFixed(2)} (${cur})`, 14, 34);
                doc.text(`Descrizione: ${expense.description}`, 14, 40);

                doc.setDrawColor(203, 213, 225);
                doc.line(14, 44, pageWidth - 14, 44);

                const topMargin = 48;
                const margin = 15;
                const maxWidth = pageWidth - (margin * 2);
                const maxHeight = pageHeight - topMargin - 15;

                let imgWidth = img.width;
                let imgHeight = img.height;

                const ratio = Math.min(maxWidth / imgWidth, maxHeight / imgHeight);
                imgWidth = imgWidth * ratio;
                imgHeight = imgHeight * ratio;

                const x = (pageWidth - imgWidth) / 2;
                const y = topMargin + (maxHeight - imgHeight) / 2;

                doc.addImage(img, 'JPEG', x, y, imgWidth, imgHeight);

                const cleanDate = expense.date.replace(/-/g, '');
                const cleanAmount = parseFloat(expense.amount).toFixed(2).replace('.', '_');
                doc.save(`scontrino_${tripDetails.lastName}_${cleanDate}_${cleanAmount}${cur}.pdf`);
            };
        }

            if (filtered.length === 0) {

                return;

            const { jsPDF } = window.jspdf;
            const doc = new jsPDF({
                orientation: 'portrait',
                unit: 'mm',
                format: 'a4'
            });

            let loadedCount = 0;
            const sortedFiltered = [...filtered].sort((a, b) => a.date.localeCompare(b.date));

            sortedFiltered.forEach((expense, index) => {
                const img = new Image();
                img.src = expense.imageSrc;
                img.onload = function() {
                    if (index > 0) {
                        doc.addPage();
                    }

                    const pageWidth = doc.internal.pageSize.getWidth();
                    const pageHeight = doc.internal.pageSize.getHeight();

                    const cur = expense.currency || 'EUR';
                    const sym = currencySymbols[cur] || cur;

                    doc.setFontSize(14);
                    doc.setTextColor(30, 41, 59);
                    doc.text(`SCONTRINO ${index + 1} DI ${sortedFiltered.length}`, 14, 15);

                    doc.setFontSize(10, 'normal');
                    doc.setTextColor(100, 116, 139);
                    doc.text(`Nome: ${tripDetails.firstName} | Cognome: ${tripDetails.lastName}`, 14, 22);
                    doc.text(`Azienda: ${tripDetails.companyName}`, 14, 28);
                    doc.text(`Data: ${new Date(expense.date).toLocaleDateString('it-IT')} | Importo: ${sym} ${parseFloat(expense.amount).toFixed(2)} (${cur})`, 14, 34);
                    doc.text(`Descrizione: ${expense.description}`, 14, 40);

                    doc.setDrawColor(203, 213, 225);
                    doc.line(14, 44, pageWidth - 14, 44);

                    const topMargin = 48;
                    const margin = 15;
                    const maxWidth = pageWidth - (margin * 2);
                    const maxHeight = pageHeight - topMargin - 15;

                    let imgWidth = img.width;
                    let imgHeight = img.height;

                    const ratio = Math.min(maxWidth / imgWidth, maxHeight / imgHeight);
                    imgWidth = imgWidth * ratio;
                    imgHeight = imgHeight * ratio;

                    const x = (pageWidth - imgWidth) / 2;
                    const y = topMargin + (maxHeight - imgHeight) / 2;

                    doc.addImage(img, 'JPEG', x, y, imgWidth, imgHeight);
                    loadedCount++;

                    if (loadedCount === sortedFiltered.length) {
                        doc.save(`scontrini_${tripDetails.lastName}_${startDate}_al_${endDate}.pdf`);
                    }
                };
            });
        }
